% Store each page as a named macro so both draft and print modes can render them
\newcommand{\setzinepage}[2]{%
  \expandafter\gdef\csname zinepage#1\endcsname{%
    \ifnum#1=1
    #2%
    \else
    \fitimposedpage{#2}%
    \fi
  }%
}

\newcommand{\fitimposedpage}[1]{%
  \iftoggle{draftmode}{#1}{%
    \begin{minipage}[t][3.75in][t]{2.35in}\raggedright #1
    \end{minipage}%
  }%
}

% Render one panel in the final imposed layout.
\newcommand{\renderpanel}[4]{% #1: Page number, #2: Rotation angle, #3: xshift, #4: yshift
  \node[
    xshift=#3,
    yshift=#4,
    rotate=#2,
    minimum width=2.75in,
    minimum height=4.25in,
    inner sep=0.18in,
    align=center,
    anchor=center
  ] at (current page.center) {%
    \iftoggle{draftmode}{%
      \begin{minipage}[t][3.85in][t]{2.35in}
        \csname zinepage#1\endcsname
      \end{minipage}%
    }{%
      \ifnum#1=1
      \begin{minipage}[t][4.25in][t]{2.75in}
        \csname zinepage#1\endcsname
      \end{minipage}%
      \else
      \begin{minipage}[t][3.75in][t]{2.35in}
        \csname zinepage#1\endcsname
      \end{minipage}%
      \fi
    }
  };
}
